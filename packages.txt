# Lumen: packages for Arch Linux
#
# One package per line; everything after # is a comment. Packages marked
# "AUR" come from the Arch User Repository and need an AUR helper (paru/yay);
# the rest are in the official repos.
#
# Install everything that isn't commented out (see README.md, step 3):
#
#   paru -S --needed $(grep -vE '^\s*(#|$)' packages.txt | awk '{print $1}')
#
# Lines starting with "# " inside a section are optional: remove the "# " to
# include them.


# ── base ────────────────────────────────────────────────────────────────
# Needed to build AUR packages and fetch this repo.
base-devel
git

# ── compositor & session ────────────────────────────────────────────────
hyprland                    # the compositor (configured in Lua); provides start-hyprland
xorg-xwayland               # X11 apps inside Hyprland
hypridle                    # screen off, lock and suspend after idle
hyprlock                    # lock screen (also shown on every login)
hyprsunset                  # night light, driven from the pill
hyprpolkitagent             # password prompts for GUI apps
polkit
xdg-desktop-portal-hyprland # screen sharing and screenshots for apps
xdg-desktop-portal-gtk      # file pickers and settings for GTK apps
xdg-utils                   # xdg-open (links and files from the pill)
xdg-user-dirs               # ~/Pictures, ~/Videos and friends

# ── audio ───────────────────────────────────────────────────────────────
pipewire
pipewire-alsa
pipewire-pulse
pavucontrol

wireplumber                 # session manager, wpctl (volume keys)

bluez-utils

ddcutil                     # brightness of external monitors over DDC/CI

quickshell                  # the whole desktop shell
awww                        # wallpaper daemon (still images)
mpvpaper                    # AUR: video wallpapers
mpv                         # player behind mpvpaper

gpu-screen-recorder         # screen recorder page
python                      # wallcolors.py: the palette engine; the dictation service
jq                          # scripts read the pill's settings
curl                        # weather, AutoEQ headphone list, dictation model download
cliphist                    # clipboard history
wl-clipboard                # wl-copy / wl-paste
wtype                       # the emoji picker and dictation type text
libnotify                   # notify-send
libarchive                  # bsdtar: the app installer unpacks archives
cava                        # music visualiser and dictation mic bars in the resting pill
libpulse                    # pactl: dictation finds the default mic
zenity                      # file picker: recorder folder, vocabulary import/export
xorg-xrandr                 # XWayland primary monitor after a display swap
hyprshot                    # Print: region screenshot
hyprpicker                  # Super+Shift+C: colour picker
